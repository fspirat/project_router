# Панель LuCI роутера через туннель. HTTPS-часть (listen 443, ssl_*) дописал certbot на сервере.
server {
    listen 80;
    server_name luci.example.org;    # свой LUCI_DOMAIN

    location / {
        auth_basic "Router";
        auth_basic_user_file /etc/nginx/.htpasswd_router;
        proxy_pass http://127.0.0.1:8081;
        proxy_set_header Host $host;
        proxy_set_header Authorization "";
        proxy_read_timeout 120s;
    }
}
